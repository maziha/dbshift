        when STATUS=1 then 'DA'
        else 'NU'

as flag_aprobat,
    dataintrarestatus as data_status_actual,
    data_Acord as data_disbursare,
case 
    when data_Acord is not null and status=1 then 'DA'
    else 'NU'
end 
as flag_disbursat,
data_Acord as data_disbursare,
case 
    when substr(retailer,1,4)='1111' then 'RBRO'
    when substr(retailer,1,4)='2222' then 'Lugera'

as canal_vanzare
from cereri_raport_mv

    F.when(F.col("level") == 1, 'Total Prescoring')
     .when(F.col("level") == 2, 'Aprobate Prescoring')
     .when(F.col("level") == 3, 'Conversie Cereri')
     .when(F.col("level") == 4, 'Aprobate Cereri')
     .when(F.col("level") == 5, 'Disbursate')
)

    F.when(F.col("level") == 1, F.lit(float(presc_total or 0)))
     .when(F.col("level") == 2, F.lit(float(aprob_presc or 0)))
     .when(F.col("level") == 3, F.lit(float(conversie_cereri or 0)))
     .when(F.col("level") == 4, F.lit(float(aprob_comm or 0)))
     .when(F.col("level") == 5, F.lit(float(disbursate or 0)))
)
