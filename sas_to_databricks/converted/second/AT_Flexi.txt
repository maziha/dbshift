def generate_column_list(var_name, num_months, start_date_str):
    start_date = datetime.strptime(start_date_str, "%Y-%m-%d")
    column_list = []
    for b in range(1, num_months + 1):
        target_date = start_date + relativedelta(months=b)
        year = target_date.year
        month_val = target_date.month
        if month_val < 10:
            column_list.append(f"b.{var_name}_{year}_0{month_val}")
        else:
            column_list.append(f"b.{var_name}_{year}_{month_val}")
    return " ," + " ,".join(column_list)
