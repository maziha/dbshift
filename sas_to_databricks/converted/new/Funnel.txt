df_stats = spark.sql("""

        tip_flux,
        luna_an_presc,

        SUM(CASE WHEN flag_disbursat = 'DA' THEN 1 ELSE 0 END) / COUNT(*) AS rata_disbursare,

# Due to the procedural nature of the SAS data step with retain, we collect the small dataframe to the driver
# and generate the polygon data using pandas/python loops. This is efficient for small datasets.
funnel2_pd = df_funnel2.toPandas()
polygon_data = []
thk = 0.48
off = 0.25

if not funnel2_pd.empty:
    # First record
    first_row = funnel2_pd.iloc[0]
    prevLow, prevHigh, prevLevel, prevValue = first_row['low'], first_row['high'], first_row['level'], first_row['check_values']
    polygon_data.append({'x': prevLow, 'y': prevLevel - thk, 'id': prevLevel, 'val': prevValue, 'yval': prevLevel - off, 'VALUE': first_row['value'], 'display_value': first_row['display_value']})
    polygon_data.append({'x': prevHigh, 'y': prevLevel - thk, 'id': prevLevel, 'val': prevValue, 'yval': prevLevel - off, 'VALUE': first_row['value'], 'display_value': first_row['display_value']})

    # Middle records
    for i in range(1, len(funnel2_pd)):
        row = funnel2_pd.iloc[i]
