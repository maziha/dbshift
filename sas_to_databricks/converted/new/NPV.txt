import pyspark.sql.functions as F
from pyspark.sql.window import Window
from pyspark.sql.types import StructType, StructField, StringType, DoubleType, IntegerType, LongType, DateType
from functools import reduce

def rest_payment_sched_new(input_data_macro_df, spark):
    
    max_counter_row = input_data_macro_df.agg(F.max("counter")).collect()
    max_counter = max_counter_row[0][0] if max_counter_row and max_counter_row[0][0] is not None else 0

    print(max_counter)

    # Schema for the table that will be built iteratively
    npv_base_all_schema = StructType([
        StructField("wo_date_1", DateType(), True),
        StructField("wo_date_2", DateType(), True),
        StructField("wo_date_3", DateType(), True),
        StructField("wo_date_4", DateType(), True),
        StructField("Write_off_1", DoubleType(), True),
        StructField("Write_off_2", DoubleType(), True),
        StructField("Write_off_3", DoubleType(), True),
        StructField("Write_off_4", DoubleType(), True),
        StructField("cod_Valuta", StringType(), True),
        StructField("cod_Valuta_ini", StringType(), True),
        StructField("lnnote_old", LongType(), True),
        StructField("lnnote", LongType(), True),
        StructField("metoda", StringType(), True),
        StructField("eir_final", DoubleType(), True),
        StructField("bal_rest_all", DoubleType(), True),
        StructField("data_restructurarii", DateType(), True),
        StructField("Nr_grafic", IntegerType(), True),
        StructField("max_nr_rest", IntegerType(), True),
        StructField("cif_client", LongType(), True),
        StructField("npv", DoubleType(), True),
        StructField("NPV_delta", DoubleType(), True),
        StructField("NPV_delta_pct", DoubleType(), True)
    ])
    df_npv_base_all = spark.createDataFrame([], npv_base_all_schema)
    
    for n in range(1, int(max_counter) + 1):
        print(f"N={n}")

        df_filtered_n = input_data_macro_df.filter(F.col("counter") == n)
        
        agg_result = df_filtered_n.agg(
            F.max("Nr_grafic_tot"),
            F.max("Nr_rest")
        ).first()

        max_nr_grafic_rest = agg_result[0] if agg_result[0] is not None else 0
        max_nr_rest = agg_result[1] if agg_result[1] is not None else 0
        
        if max_nr_grafic_rest > 99:
            max_nr_grafic_rest1 = str(int(max_nr_grafic_rest))
        elif max_nr_grafic_rest > 9:
            max_nr_grafic_rest1 = str(int(max_nr_grafic_rest))
        else:
            max_nr_grafic_rest1 = str(int(max_nr_grafic_rest))

        # print(max_nr_grafic_rest1)
        
        detail_dfs = []
        for j in range(1, int(max_nr_grafic_rest1) + 1):
            
            df_temp = df_filtered_n.filter(F.col("nr_grafic_tot") == j)
            
            if j == int(max_nr_grafic_rest1):
                # Add 1 period to last trench
                df_temp = df_temp.withColumn("l", F.explode(F.sequence(F.lit(1), F.col("plati_in_grafic_tot") + 1)))
            else:
                df_temp = df_temp.withColumn("l", F.explode(F.sequence(F.lit(1), F.col("plati_in_grafic_tot"))))

            detail_dfs.append(df_temp)

        if not detail_dfs:
            continue

        df_rest_sched_all = reduce(lambda df1, df2: df1.unionByName(df2), detail_dfs)

        window_spec_n = Window.partitionBy("lnnote", "nr_rest").orderBy("l").rowsBetween(Window.unboundedPreceding, Window.currentRow)
        window_spec_npv = Window.partitionBy("lnnote", "nr_rest").orderBy("l").rowsBetween(Window.unboundedPreceding, Window.currentRow)

        df_rest_sched_all = df_rest_sched_all.withColumn("n", F.row_number().over(Window.partitionBy("lnnote", "nr_rest").orderBy("l")))
        df_rest_sched_all = df_rest_sched_all.withColumn("Nr_grafic", F.lit(int(max_nr_grafic_rest1)))
        df_rest_sched_all = df_rest_sched_all.withColumn("Max_nr_rest", F.lit(max_nr_rest))
        df_rest_sched_all = df_rest_sched_all.withColumn("r", F.col("EIR_final") / 1200)
        df_rest_sched_all = df_rest_sched_all.withColumn("disc_rate", F.pow((F.lit(1) + F.col("r")), F.col("n")))
        df_rest_sched_all = df_rest_sched_all.withColumn("PV", F.col("suma_plata_all") / F.col("disc_rate"))
        
        df_rest_sched_all = df_rest_sched_all.withColumn("npv_calc", F.sum("PV").over(window_spec_npv))
        df_rest_sched_all = df_rest_sched_all.withColumn("npv", F.when(F.col("n") == 1, 0).otherwise(F.col("npv_calc")))
        
        df_rest_sched_all = df_rest_sched_all.withColumn("Due_date", F.expr("add_months(DATA_PRIMEI_PLATI, l - 1)"))

        # Replicating NPV+PV;
        window_spec_running_npv = Window.partitionBy("lnnote", "nr_rest").orderBy("l").rowsBetween(Window.unboundedPreceding, Window.currentRow)
        df_rest_sched_all = df_rest_sched_all.withColumn("npv", F.sum("PV").over(window_spec_running_npv))

        window_spec_last = Window.partitionBy("lnnote").orderBy(F.col("Due_date").desc())
        
        df_rest_sched_all_1 = df_rest_sched_all.withColumn("rn", F.row_number().over(window_spec_last))
        df_rest_sched_all_1 = df_rest_sched_all_1.filter(F.col("rn") == 1)
        
        df_rest_sched_all_1 = df_rest_sched_all_1.withColumn("NPV_delta", F.col("bal_rest_all") - F.col("npv"))
        df_rest_sched_all_1 = df_rest_sched_all_1.withColumn("NPV_delta_pct", (F.col("bal_rest_all") - F.col("npv")) / F.col("bal_rest_all"))
        
        df_rest_sched_all_1 = df_rest_sched_all_1.select(
            "wo_date_1", "wo_date_2", "wo_date_3", "wo_date_4",
            "Write_off_1", "Write_off_2", "Write_off_3", "Write_off_4",
            "cod_Valuta", "cod_Valuta_ini", "lnnote_old", "lnnote", "metoda", "eir_final",
            "bal_rest_all", "data_restructurarii", "Nr_grafic", "max_nr_rest", "cif_client", "npv",
            "NPV_delta", "NPV_delta_pct"
        )
        
        df_npv_base_all = df_npv_base_all.unionByName(df_rest_sched_all_1)

    return df_npv_base_all

        df_cons_data_sql = spark.sql("""
