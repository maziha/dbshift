def campuri_param(var):
    b_var_list = []
    start_date = datetime.strptime(begin_dpd, "%Y-%m-%d")
    for b in range(1, numar_luni_dpd + 1):
        target_date = start_date + relativedelta(months=b)
        year = target_date.year
        month_val = target_date.month
        if month_val < 10:
            b_var_list.append(f"b.{var}_{year}_0{month_val}")
        else:
            b_var_list.append(f"b.{var}_{year}_{month_val}")
    
    b_var_str = " ," + " ,".join(b_var_list)
    print(f"b{var}={b_var_str}")
    return b_var_str

def get_segment(df, seg_col_name):
    df_map_var = spark.table("DanM.SC_FLEXI_MAP_var").filter(F.col("SC") == seg_col_name).withColumnRenamed("Segment", f"V_{seg_col_name}")
    df_joined = df.join(df_map_var, df[seg_col_name] == df_map_var["SCOR"], "left").drop("SC", "SCOR")
    df_joined.groupBy(f"V_{seg_col_name}").count().show()
    return df_joined
