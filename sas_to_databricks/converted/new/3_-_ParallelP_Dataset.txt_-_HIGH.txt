# Date filter update logic
# This translates the PROC SQL EXECUTE (INSERT ... SELECT) BY Teradata.
# It reads the necessary data, calculates the new row, and appends it.
print("Updating fcst_update_dtinf control table...")
sql_query_for_insert = """
SELECT Current_Date AS upd_date, Max(a.startDate) AS startDate, Max(wk_end_dt) AS endDate, Count(DISTINCT wk_end_dt) AS nweeks
FROM
(
    SELECT max(endDate) + 1 AS startDate
    FROM fcst_update_dtinf AS a
) AS a,
vprod_dim.dt_inf AS b
WHERE wk_end_dt BETWEEN startDate AND Current_Date
"""
try:
    df_to_insert = spark.read \
        .format("jdbc") \
        .option("url", jdbc_url) \
        .option("dbtable", f"({sql_query_for_insert}) as subq") \
        .option("user", connection_properties["user"]) \
        .option("password", connection_properties["password"]) \
        .load()

    df_to_insert.write \
        .format("jdbc") \
        .option("url", jdbc_url) \
        .option("dbtable", "fcst_update_dtinf") \
        .option("user", connection_properties["user"]) \
        .option("password", connection_properties["password"]) \
        .mode("append") \
        .save()
    print("Successfully inserted new date range into fcst_update_dtinf.")
except Exception as e:
    print(f"An error occurred during the control table update: {e}")

    .format("jdbc") \
    .option("url", jdbc_url) \

    .option("user", connection_properties["user"]) \
    .option("password", connection_properties["password"]) \
    .load()

    F.max("startDate").alias("max_start_date"),
    F.max("endDate").alias("max_end_date")
)

    F.concat(F.lit("'"), F.date_format(F.col("max_start_date"), "yyyy-MM-dd"), F.lit("'")).alias("st_dt"),
    F.concat(F.lit("'"), F.date_format(F.col("max_end_date"), "yyyy-MM-dd"), F.lit("'")).alias("end_dt"),
    F.upper(F.date_format(F.col("max_start_date"), "ddMMMyyyy")).alias("st_dt2"),
