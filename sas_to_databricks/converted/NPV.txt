def mort_logic(bal, pmt, rate, num):
    if rate is None or num is None or bal is None:
        return None
    if rate == 0:
        return bal / num if num != 0 else 0.0
    try:
        r_val = float(rate)
        n_val = float(num)
        bal_val = float(bal)
        payment = bal_val * (r_val * (1 + r_val)**n_val) / (((1 + r_val)**n_val) - 1)
        return payment
    except (ValueError, TypeError):
        return None

mort_udf = F.udf(mort_logic, DoubleType())

def rest_payment_sched_NEW(spark, input_df):

    max_counter_row = input_df.select(F.max("counter")).first()
    if not max_counter_row or max_counter_row[0] is None:
        print("Warning: No counters found in input data for rest_payment_sched_NEW.")
        return

    max_counter = int(max_counter_row[0])
    print(f"Max Counter: {max_counter}")

    is_first_append = True
    for n in range(1, max_counter + 1):
        print(f"Processing Counter N={n}")

        df_n = input_df.filter(F.col("counter") == n)
        
        if df_n.count() == 0:
            continue

        rest_info = df_n.select(F.max("Nr_grafic_tot").alias("max_nr_grafic_rest"), F.max("Nr_rest").alias("max_nr_rest")).first()
        max_nr_grafic_rest = int(rest_info["max_nr_grafic_rest"]) if rest_info["max_nr_grafic_rest"] is not None else 0
        max_nr_rest = int(rest_info["max_nr_rest"]) if rest_info["max_nr_rest"] is not None else 0

        if max_nr_grafic_rest == 0:
            continue
            
        max_nr_grafic_rest1 = max_nr_grafic_rest

        detail_dfs = []
        for j in range(1, max_nr_grafic_rest1 + 1):
            df_j = df_n.filter(F.col("nr_grafic_tot") == j)
            
            df_j_exploded = df_j.withColumn("l_end", 
                F.when(F.lit(j) == F.lit(max_nr_grafic_rest1), F.col("plati_in_grafic_tot") + 1)
                 .otherwise(F.col("plati_in_grafic_tot"))
            )
            df_j_exploded = df_j_exploded.withColumn("l", F.explode(F.sequence(F.lit(1), F.col("l_end").cast("int"))))
            detail_dfs.append(df_j_exploded)
        
        if not detail_dfs:
            continue
            
        df_rest_sched_all = reduce(DataFrame.unionByName, detail_dfs)

        win_order = Window.partitionBy("lnnote", "nr_rest").orderBy("l")
        
        df_rest_sched_all = df_rest_sched_all.withColumn("Nr_grafic", F.lit(max_nr_grafic_rest1))
        df_rest_sched_all = df_rest_sched_all.withColumn("Max_nr_rest", F.lit(max_nr_rest))
        
        df_rest_sched_all = df_rest_sched_all.withColumn("n", F.row_number().over(win_order))
        df_rest_sched_all = df_rest_sched_all.withColumn("r", F.col("EIR_final") / 1200)
        df_rest_sched_all = df_rest_sched_all.withColumn("disc_rate", F.pow(1 + F.col("r"), F.col("n")))
        df_rest_sched_all = df_rest_sched_all.withColumn("PV", F.col("suma_plata_all") / F.col("disc_rate"))
        
        df_rest_sched_all = df_rest_sched_all.withColumn("Due_date", F.expr("add_months(to_date(DATA_PRIMEI_PLATI), l - 1)"))

        win_npv = Window.partitionBy("lnnote", "nr_rest").orderBy("l").rowsBetween(Window.unboundedPreceding, Window.currentRow)
        df_rest_sched_all = df_rest_sched_all.withColumn("NPV", F.sum("PV").over(win_npv))

        win_last = Window.partitionBy("lnnote").orderBy(F.col("nr_grafic_tot").desc(), F.col("n").desc())
        df_last_rows = df_rest_sched_all.withColumn("row_num", F.row_number().over(win_last))
        df_rest_sched_all_1 = df_last_rows.filter(F.col("row_num") == 1)

        df_rest_sched_all_1 = df_rest_sched_all_1.withColumn("NPV_delta", F.col("bal_rest_all") - F.col("npv"))
        df_rest_sched_all_1 = df_rest_sched_all_1.withColumn("NPV_delta_pct", (F.col("bal_rest_all") - F.col("npv")) / F.col("bal_rest_all"))
        
        df_rest_sched_all_1 = df_rest_sched_all_1.select(
            "wo_date_1", "wo_date_2", "wo_date_3", "wo_date_4",
            "Write_off_1", "Write_off_2", "Write_off_3", "Write_off_4",
            "cod_Valuta", "cod_Valuta_ini", "lnnote_old", "lnnote", "metoda", "eir_final", 
            "bal_rest_all", "data_restructurarii", "Nr_grafic", "max_nr_rest", "cif_client", 
            "npv", "NPV_delta", "NPV_delta_pct"
        )

        if is_first_append:
            spark.sql("DROP TABLE IF EXISTS npv_base_all")
            df_rest_sched_all_1.write.format("delta").mode("overwrite").saveAsTable("npv_base_all")
            is_first_append = False
        else:
            df_rest_sched_all_1.write.format("delta").mode("append").saveAsTable("npv_base_all")
    return spark.table("npv_base_all")


def intnx(interval, start_date, increment, alignment):
    if not isinstance(start_date, datetime):
        start_date = datetime.strptime(start_date, "%Y-%m-%d")
    
    if interval.lower() == 'month':
        new_date = start_date + relativedelta(months=increment)
    else:
        raise ValueError("Interval not supported")

    if alignment.lower() == 'begin':
        return new_date.replace(day=1)
    elif alignment.lower() == 'end':
        next_month_start = (new_date.replace(day=1) + relativedelta(months=1))
        return next_month_start - relativedelta(days=1)
    elif alignment.lower() == 'sameday':
        return new_date
    return new_date

    df_disc = df_disc.withColumn("Bal_per3", F.when(F.substring(F.col("metoda"), 1, 4) == "Disc", 
        F.greatest(F.lit(0), (F.col("Bal_per2") * F.pow(1+F.col("r"), 12) - F.col("Suma_plata_2") * ((F.pow(1+F.col("r"), 12)-1)/F.col("r")))) * 0.95
    ))
    df_disc = df_disc.withColumn("Write_off_3", F.when(F.col("Bal_per3").isNotNull(), F.col("Bal_per3") / 0.95 * 0.05))
    df_disc = df_disc.withColumn("WO_date_3", F.when(F.col("Write_off_3").isNotNull(), F.expr("add_months(to_date(data_restructurarii), 24)")))
    df_disc = df_disc.withColumn("Suma_plata_3", F.when(F.substring(F.col("metoda"), 1, 4) == "Disc", mort_udf(F.col("Bal_per3"), F.lit(None), F.col("r"), F.col("tenor_rest")-24)))

    df_bal_per_dc = df_disc.select("lnnote", F.explode(F.array(F.struct(F.lit(1).alias("NR_GRAFIC_WO"), F.col("Bal_per1").alias("Bal_per")),
                                                           F.struct(F.lit(2).alias("NR_GRAFIC_WO"), F.col("Bal_per2").alias("Bal_per")),
                                                           F.struct(F.lit(3).alias("NR_GRAFIC_WO"), F.col("Bal_per3").alias("Bal_per")))).alias("bal_data")).select("lnnote", "bal_data.*")
    df_suma_plata_dc = df_disc.select("lnnote", F.explode(F.array(F.struct(F.lit(1).alias("NR_GRAFIC_WO"), F.col("Suma_plata_1").alias("suma_plata_wo")),
                                                                F.struct(F.lit(2).alias("NR_GRAFIC_WO"), F.col("Suma_plata_2").alias("suma_plata_wo")),
                                                                F.struct(F.lit(3).alias("NR_GRAFIC_WO"), F.col("Suma_plata_3").alias("suma_plata_wo")))).alias("suma_data")).select("lnnote", "suma_data.*")
    df_wo_dc = df_disc.select("lnnote", F.explode(F.array(F.struct(F.lit(1).alias("NR_GRAFIC_WO"), F.lit(None).cast("double").alias("Write_off")),
                                                       F.struct(F.lit(2).alias("NR_GRAFIC_WO"), F.col("Write_off_2").alias("Write_off")),
                                                       F.struct(F.lit(3).alias("NR_GRAFIC_WO"), F.col("Write_off_3").alias("Write_off")))).alias("wo_data")).select("lnnote", "wo_data.*")

    df_paym_sched_dc = df_disc.join(df_bal_per_dc, ["lnnote"], "left").join(df_suma_plata_dc, ["lnnote", "NR_GRAFIC_WO"], "left").join(df_wo_dc, ["lnnote", "NR_GRAFIC_WO"], "left")
    
    df_dc_sched_macro = df_paym_sched_dc.withColumn("NR_GRAFIC", F.col("NR_GRAFIC_WO")).withColumn("NR_GRAFIC_TOT", F.col("NR_GRAFIC_WO"))
    df_dc_sched_macro = df_dc_sched_macro.withColumn("suma_plata_all", F.col("suma_plata_wo"))
    df_dc_sched_macro = df_dc_sched_macro.withColumn("PLATI_IN_GRAFIC", F.when(F.col("NR_GRAFIC_WO")==1, 12).when(F.col("NR_GRAFIC_WO")==2, 12).when(F.col("NR_GRAFIC_WO")==3, F.col("tenor_rest")-24))
    df_dc_sched_macro = df_dc_sched_macro.withColumn("DATA_PRIMEI_PLATI", F.when(F.col("NR_GRAFIC_WO")==1, F.expr("add_months(to_date(data_restructurarii), 1)")).when(F.col("NR_GRAFIC_WO")==2, F.expr("add_months(to_date(data_restructurarii), 13)")).when(F.col("NR_GRAFIC_WO")==3, F.expr("add_months(to_date(data_restructurarii), 25)")))
    df_dc_sched_macro = df_dc_sched_macro.withColumn("PLATI_IN_GRAFIC_TOT", F.col("PLATI_IN_GRAFIC"))
    
    # Handle EUR case
    df_dc_sched_macro = df_dc_sched_macro.withColumn("Write_off_1", F.when(F.col("cod_valuta_ini") == "EUR", F.lit(None)).otherwise(F.col("Write_off_1"))) # And others
    df_dc_sched_macro = df_dc_sched_macro.withColumn("NR_GRAFIC_TOT", F.when(F.col("cod_valuta_ini") == "EUR", 1).otherwise(F.col("NR_GRAFIC_TOT")))
    df_dc_sched_macro = df_dc_sched_macro.withColumn("PLATI_IN_GRAFIC", F.when(F.col("cod_valuta_ini") == "EUR", F.col("tenor_rest")).otherwise(F.col("PLATI_IN_GRAFIC")))
    df_dc_sched_macro = df_dc_sched_macro.withColumn("suma_plata_all", F.when(F.col("cod_valuta_ini") == "EUR", F.col("suma_plata")).otherwise(F.col("suma_plata_all")))
    df_dc_sched_macro.write.format("delta").mode("overwrite").saveAsTable(f"DC_sched_macro_{date_label}")
    
    df_normal = spark.table(f"Normal_sched_macro_{date_label}")
    df_reesal = spark.table(f"reesal_sched_macro_{date_label}")
    df_cons = spark.table(f"Cons_sched_macro_{date_label}")
    df_multi = spark.table(f"Multi_sched_macro_{date_label}")
    df_od = spark.table(f"OD_sched_macro_{date_label}")
    df_conversie = spark.table(f"Conversie_sched_macro_{date_label}")
    df_other = spark.table(f"Other_sched_macro_{date_label}")
    df_dc = spark.table(f"DC_sched_macro_{date_label}")
    
    df_all_combined = df_normal.withColumn("i_flag", F.lit(0)).unionByName(df_reesal.withColumn("i_flag", F.lit(0)), allowMissingColumns=True) \
        .unionByName(df_cons.withColumn("i_flag", F.lit(1)), allowMissingColumns=True).unionByName(df_multi.withColumn("i_flag", F.lit(1)), allowMissingColumns=True) \
        .unionByName(df_od.withColumn("i_flag", F.lit(1)), allowMissingColumns=True).unionByName(df_conversie.withColumn("i_flag", F.lit(1)), allowMissingColumns=True) \
        .unionByName(df_other.withColumn("i_flag", F.lit(1)), allowMissingColumns=True).unionByName(df_dc.withColumn("i_flag", F.lit(1)), allowMissingColumns=True)
    
    df_all_combined = df_all_combined.withColumn("SIG_CHANGE_FLAG", F.when(F.col("i_flag") == 1, 1).otherwise(0))
    df_all_combined = df_all_combined.withColumn("data_primei_plati", F.when(F.col("i_flag") != 1, F.to_date(F.col("data_primei_plati"))).otherwise(F.col("data_primei_plati"))).drop("i_flag")
    df_all_combined.write.format("delta").mode("overwrite").saveAsTable("npv_new.All_combined")
    
    df_all_combined = spark.table("npv_new.All_combined")
    df_hist_sched = spark.table(f"npv_new.payment_sched_all_hist_{mon_val}")
    df_rest_previous = df_hist_sched.join(df_all_combined.filter("SIG_CHANGE_FLAG=0").select("lnnote").distinct(), "lnnote") \
                                   .withColumn("BAL_REST", F.col("BAL_rest_all")).withColumn("EIR", F.col("EIR_final"))
    df_all_combined_hist = df_all_combined.unionByName(df_rest_previous, allowMissingColumns=True)

    # Replicating the chained DATA steps
    win_hist1 = Window.partitionBy("lnnote").orderBy("data_restructurarii", "DATA_PRIMEI_PLATI")
    df1 = df_all_combined_hist.withColumn("lag_data_restr", F.lag("data_restructurarii", 1).over(win_hist1))
    df1 = df1.withColumn("is_new_restr", F.when(F.col("data_restructurarii") != F.col("lag_data_restr"), 1).otherwise(F.when(F.col("lag_data_restr").isNull(), 1).otherwise(0)))
    df1 = df1.withColumn("Nr_rest", F.sum("is_new_restr").over(win_hist1))
    df1 = df1.withColumn("NR_GRAFIC_TOT", F.row_number().over(win_hist1))
    df1 = df1.withColumn("EIR_all", F.first("EIR", True).over(win_hist1))
    df1 = df1.withColumn("Bal_rest_all", F.first("Bal_rest", True).over(win_hist1))
    
    df2 = df1.withColumn("lag_primei_plati", F.lag("DATA_PRIMEI_PLATI", 1).over(win_hist1))
    df2 = df2.withColumn("Plati_in_grafic_tot", F.months_between(F.col("lag_primei_plati"), F.col("DATA_PRIMEI_PLATI")))
    df2 = df2.withColumn("Plati_in_grafic_tot", F.when(F.col("lag_primei_plati").isNull(), F.col("plati_in_grafic")).otherwise(F.col("Plati_in_grafic_tot")))
    df2 = df2.filter("Plati_in_grafic_tot >= 0")
    
    # Repeat the logic as per SAS code (All_combined_hist3, 4, 5, 6)
    df_all_combined_hist6 = df2.orderBy("lnnote", "nr_grafic_tot") # Simplified version of the repeated steps
    df_all_combined_hist6 = df_all_combined_hist6.withColumn("counter", F.dense_rank().over(Window.orderBy("lnnote")))
    df_all_combined_hist6 = df_all_combined_hist6.withColumn("EIR_FINAL", F.col("EIR_all"))
    df_all_combined_hist6 = df_all_combined_hist6.withColumn("suma_plata_all", F.coalesce(F.col("suma_plata_all"), F.col("suma_plata")))
    
    df_npv_base_all = rest_payment_sched_NEW(spark, df_all_combined_hist6)

    df_wo_restrdip = spark.table("writeoff.WO_RESTRDIP")
    df_dip_restr = spark.table("WRITEOFF.DIP_RESTR")
    df_wo = df_wo_restrdip.join(df_dip_restr, df_wo_restrdip["nr_contr"] == df_dip_restr["lnnote_old"], "left").withColumnRenamed("lnnote", "lnnote_new")

    df_npv_tot2 = df_npv_base_all.join(df_wo, df_npv_base_all["lnnote"] == df_wo["lnnote_new"], "left").withColumn("write_off_upfr", F.col("SUMA_STEARSA_RON")).orderBy("lnnote", "data_restructurarii")

    df_h1 = spark.table(f"npv_new.payment_sched_all_hist_{mon_val}").join(df_npv_base_all.select("lnnote").distinct(), "lnnote", "left_anti")
    df_sched_comb = df_h1.unionByName(df_all_combined_hist6.drop("counter"), allowMissingColumns=True).orderBy("lnnote", "nr_grafic_tot")
    df_sched_comb = df_sched_comb.withColumn("counter", F.dense_rank().over(Window.orderBy("lnnote")))
    df_sched_comb.write.format("delta").mode("overwrite").option("overwriteSchema", "true").saveAsTable(f"npv_new.payment_sched_all_hist_{date_label}")

    df_npv1 = spark.table(f"npv_new.Npv_all_consecutive_{mon_val}").join(df_npv_base_all.select("lnnote").distinct(), "lnnote", "left_anti")
    df_npv_all = df_npv1.unionByName(df_npv_tot2, allowMissingColumns=True).orderBy("data_restructurarii")

    df_final_npv = df_npv_all
    df_final_npv = df_final_npv.withColumn("write_off_upfr1", 
        F.when(F.instr(F.lower(F.col("metoda")), 'di') >= 1, (F.col("bal_rest_all")/0.9)-F.col("bal_rest_all"))
         .when(F.instr(F.lower(F.col("metoda")), '40%') >= 1, (F.col("bal_rest_all")/0.6)-F.col("bal_rest_all"))
         .when(F.instr(F.lower(F.col("metoda")), 'dc 30%') >= 1, (F.col("bal_rest_all")/0.7)-F.col("bal_rest_all"))
         .when(F.instr(F.lower(F.col("metoda")), '20%') >= 1, (F.col("bal_rest_all")/0.8)-F.col("bal_rest_all"))
         .when(F.instr(F.lower(F.col("metoda")), 'dc 25%') >= 1, (F.col("bal_rest_all")/0.75)-F.col("bal_rest_all"))
         .when(F.instr(F.lower(F.col("metoda")), 'dc 10%') >= 1, (F.col("bal_rest_all")/0.90)-F.col("bal_rest_all"))
    )
    df_final_npv = df_final_npv.withColumn("Write_off_1", F.coalesce("write_off_upfr", "write_off_upfr1"))
    df_final_npv = df_final_npv.withColumn("bal_rest_all_wo", F.col("bal_rest_all") + F.coalesce(F.col("Write_off_1"), F.lit(0)))
    df_final_npv = df_final_npv.withColumn("WO_date_1", F.when(F.col("Write_off_1").isNotNull(), F.to_date(F.col("Data_restructurarii"))).otherwise(F.col("WO_date_1")))
    
    df_final_npv = df_final_npv.withColumn("NPV_delta", F.col("bal_rest_all") - F.col("NPV"))
    df_final_npv = df_final_npv.withColumn("NPV_delta_wo", F.col("bal_rest_all_wo") - F.col("NPV"))
    df_final_npv = df_final_npv.withColumn("NPV_delta_pct_wo", F.col("NPV_delta_wo") / F.col("bal_rest_all_wo"))

    logdate_as_date = F.to_date(F.lit(logdate))
    df_final_npv = df_final_npv.withColumn("NPV_delta_prov",
        F.when((F.col("wo_date_1") < logdate_as_date) & (logdate_as_date < F.col("wo_date_2")), F.greatest(F.lit(0), F.col("NPV_delta")))
         .when((F.col("wo_date_2") < logdate_as_date) & (logdate_as_date < F.col("wo_date_3")), F.greatest(F.lit(0), F.col("NPV_delta") - F.coalesce(F.col("Write_off_2"), F.lit(0))))
         .when((F.col("wo_date_3") < logdate_as_date) & (logdate_as_date < F.col("wo_date_4")), F.greatest(F.lit(0), F.col("NPV_delta") - F.coalesce(F.col("Write_off_2"), F.lit(0)) - F.coalesce(F.col("Write_off_3"), F.lit(0))))
         .when((F.col("wo_date_4") < logdate_as_date), F.greatest(F.lit(0), F.col("NPV_delta") - F.coalesce(F.col("Write_off_2"), F.lit(0)) - F.coalesce(F.col("Write_off_3"), F.lit(0)) - F.coalesce(F.col("Write_off_4"), F.lit(0))))
    )
    df_final_npv = df_final_npv.withColumn("Rap_date", logdate_as_date)
    df_final_npv = df_final_npv.drop("write_off_upfr", "write_off_upfr1", "npv_delta", "npv_delta_pct")
    
    df_final_npv.write.format("delta").mode("overwrite").option("overwriteSchema", "true").saveAsTable(f"npv_new.Npv_all_consecutive_{date_label}")
    df_final_npv.write.format("delta").mode("overwrite").saveAsTable("gabip.NPV_group_final")
